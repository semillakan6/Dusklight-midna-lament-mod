cmake_minimum_required(VERSION 3.26)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
project(midna_lament_battle VERSION 1.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The game headers use __cplusplus to decide whether nullptr needs emulation.
# MSVC otherwise reports C++98 even in /std:c++20 mode.
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/Zc:__cplusplus>")
set(DUSKLIGHT_VERSION "40457c6adb381928e4b5fef6ed459ed291edd5e2")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/FetchDusklight.cmake")
add_subdirectory("${DUSKLIGHT_DIR}/sdk" dusklight-sdk EXCLUDE_FROM_ALL)
configure_file(mod.json.in "${CMAKE_CURRENT_BINARY_DIR}/mod.json" @ONLY NEWLINE_STYLE LF)
find_package(Threads REQUIRED)
add_mod(midna_lament_battle FEATURES game
  SOURCES src/mod.cpp MOD_JSON "${CMAKE_CURRENT_BINARY_DIR}/mod.json" OVERLAY_DIR overlay)
target_link_libraries(midna_lament_battle PRIVATE Threads::Threads)
option(LAMENT_RUNTIME_PROBE "Build audio-only test injection (never ship)" OFF)
if(LAMENT_RUNTIME_PROBE)
  target_compile_definitions(midna_lament_battle PRIVATE LAMENT_RUNTIME_PROBE=1)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(controller_tests tests/controller_tests.cpp)
  add_test(NAME controller_tests COMMAND controller_tests)
endif()

option(LAMENT_LIFECYCLE_PROBE "Build disposable-profile lifecycle driver (never ship)" OFF)
if(LAMENT_LIFECYCLE_PROBE)
  add_mod(lifecycle_probe FEATURES game SOURCES tests/lifecycle_probe.cpp MOD_JSON tests/lifecycle-probe.json)
endif()

option(LAMENT_SECTION_PROBE "Build disposable story-save audio observer (never ship)" OFF)
if(LAMENT_SECTION_PROBE)
  add_mod(section_probe FEATURES game SOURCES tests/section_probe.cpp MOD_JSON tests/section-probe.json)
endif()
